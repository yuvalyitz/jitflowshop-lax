% The due-date profile of Section 5 with q_max = 4: how it changes from s_{j-1} to s_j, and when job j
% with q_j = 3 is selected. Adapted from the slide "Altered DP bounded q_max".
\begin{tikzpicture}[x=0.95cm,y=1cm]
\colorlet{prep}{red!70!black}
\tikzset{cell/.style={draw,minimum width=0.95cm,minimum height=0.6cm,inner sep=0pt,anchor=south west}}

% the profile at s_{j-1}: entry i counts the selected jobs due at s_{j-1} + i
\node[anchor=east,font=\footnotesize,align=right] at (-0.2,2.3) {profile at $s_{j-1}$};
\foreach \i/\v in {1/1, 2/0, 3/1, 4/2}
{
    \node[cell] at (\i-1,2.0) {$\v$};
    \node[font=\scriptsize] at (\i-0.5,2.8) {$\i$};
}
\draw[decorate,decoration={brace,mirror,amplitude=4pt}] (0,1.9) -- (2,1.9)
    node[midway,yshift=-11pt,font=\footnotesize] {$\delta_j$};

% the same jobs seen from s_j = s_{j-1} + delta_j: the profile moves left by delta_j = 2
\node[anchor=east,font=\footnotesize,align=right] at (1.8,0.9) {profile at $s_j$,\\$j$ passed over};
\foreach \i/\v in {1/1, 2/2, 3/0, 4/0}
    \node[cell] at (\i+1,0.6) {$\v$};
\foreach \i in {3,4}
    \draw[-{Stealth[length=2mm]},thick] (\i-0.5,1.95) -- (\i-0.5,1.25);

% selecting j with q_j = 3 puts one more job into entry 3
\node[anchor=east,font=\footnotesize,align=right] at (1.8,-0.3) {profile at $s_j$,\\$j$ selected, $q_j=3$};
\foreach \i/\v in {1/1, 2/2, 3/1, 4/0}
{
    \ifnum\i=3
        \node[cell,draw=prep,fill=red!14,line width=0.9pt] at (\i+1,-0.6) {$\v$};
    \else
        \node[cell] at (\i+1,-0.6) {$\v$};
    \fi
}
\end{tikzpicture}
